const Attendance = require('../models/Attendance');
exports.downloadCSV = async (req, res) => {
  try {
    const { class: cls, section, startDate, endDate } = req.query;
    const query = {};
    if (startDate || endDate) {
      query.date = {};
      if (startDate) query.date.$gte = new Date(startDate);
      if (endDate) query.date.$lte = new Date(endDate);
    }
    let students = [];
    if (cls || section) {
      students = await (await import('../models/Student.js')).default.find(
        cls ? { class: cls } : { section }
      );
      if (students.length) query.student = { $in: students.map(s => s._id) };
    }
    const records = await Attendance.find(query)
      .populate('student', 'name rollNumber class section')
      .lean();
    const csvHeader = 'Date,Student Name,Roll Number,Class,Section,Status\n';
    const rows = records.map(r => {
      const d = new Date(r.date).toISOString().split('T')[0];
      return `${d},${r.student.name},${r.student.rollNumber},${r.student.class},${r.student.section},${r.status}`;
    }).join('\n');
    res.header('Content-Type', 'text/csv');
    res.header('Content-Disposition', 'attachment; filename=attendance_report.csv');
    res.send(csvHeader + rows);
  } catch (err) { res.status(500).json({ error: err.message }); }
};
